        var areas = [];
        // 강원도
        var area1 = ["강릉", "강원(고성)", "대관령", "동해", "삼척", "양양", "영월", "원주", "주문진", "춘천", "태백"];

        // 울릉도

        // 충청남도
        var area8 = ["논산", "당진", "보령", "서산", "서천", "아산", "천안", "태안"];
        // 충청북도
        var area8_2 = ["소백산", "제천", "청주", "청주공항", "추풍령", "충주"];
        // 대전
        var area9 = ["대전"];
        // 세종
        var area9_2 = ["세종"];
        // 대구
        var area10 = ["대구"];
        // 전라북도
        var area11 = ["군산", "남원", "변산", "부안", "익산", "임실", "장수", "전주", "정읍"];

        $(function () {
        var areaList = $('ul.areaList');

        var monthInput = $('#monthInput');

        function ajaxRequest(location, date) {

            if(date.substring(0,4) < 2023 && location == '강원(고성)'){
                location = '고성(강원)'
            }
            if(date.substring(0,4) >= 2023 && location == '강원(고성)'){
                location = '강원고성'
            }
            if(date.substring(0,4) >= 2023 && location == '광주(경기)'){
                location = '경기광주'
            }
            if(date.substring(0,4) >= 2023 && location == '보현산'){
                location = '보현산천문대';
            }
            if(date.substring(0,4) >= 2023 && location == '소백산'){
                location = '소백산천문대';
            }
            if(date.substring(0,4) >= 2023 && location == '대덕'){
                location = '대덕전파천문대';
            }
            if(date.substring(0,4) >= 2023 && location == '제주(레이다기지)'){
                location = '제주레이더';
            }

            if(!isRunning){
                isRunning = true;

                $.ajax({
                    url : "/life/sunmoon",
                    data : {
                        location : location,
                        date : date
                    },
                    beforeSend : function(){
                        var yyyy = date.substr(0, 4);
                        var mm = date.substr(4, 2);

                        yearInput.val(yyyy);
                        monthInput.val(parseInt(mm));

                        titleMonthSpan.text(yyyy + '년 ' + parseInt(mm) + '월');
                        sunmoonTableBody.empty();

                        $('#excelLink').attr('href', '/life/sunmoon/excel?location=' + encodeURI(location) + '&date=' + date);
                    },
                    success : function(datas){
                        var sunmoon = datas.sunmoon;
                        var luncList = datas.luncList;

                        if(sunmoon.length > 0) {
                            sunmoon.forEach(function (data, i) {
                                var tr = '';

                                tr += '<tr>';
                                tr += '<td>' + data.RGON_DATE.substr(6) + '</td>';
                                tr += '<td>' + (luncList.length === 0 || luncList[i].LUNC === undefined ? '자료없음' : luncList[i].LUNC) + '</td>';
                                tr += '<td>' + formatTime(data.SUNRISE_TM) + '</td>';
                                tr += '<td>' + formatTime(data.SUN_TRANSIT_TM) + '</td>';
                                tr += '<td>' + formatTime(data.SUNSET_TM) + '</td>';
                                tr += '<td>' + formatTime(data.MOONRISE_TM) + '</td>';
                                tr += '<td>' + formatTime(data.MOON_TRANSIT_TM) + '</td>';
                                tr += '<td>' + formatTime(data.MOONSET_TM) + '</td>';
                                tr += '<td>' + formatTime(data.CIVIL_TWIL_TM_MORN) + '</td>';
                                tr += '<td>' + formatTime(data.CIVIL_TWIL_TM_EVEN) + '</td>';
                                tr += '<td>' + formatTime(data.NAUT_TWIL_TM_MORN) + '</td>';
                                tr += '<td>' + formatTime(data.NAUT_TWIL_TM_EVEN) + '</td>';
                                tr += '<td>' + formatTime(data.ASTRO_TWIL_TM_MORN) + '</td>';
                                tr += '<td>' + formatTime(data.ASTRO_TWIL_TM_EVEN) + '</td>';
                                tr += '</tr>';

                                sunmoonTableBody.append(tr);
                            });
                        } else {
                            sunmoonTableBody.append('<tr><td colspan="14">자료없음</td></tr>');
                        }
                    },
                    complete : function () {
                        isRunning = false;
                    }
                });
            }
        }

        function checkUndefined(val){
            if(val === undefined) {
                return '-';
            }
            return val;
        }

        $('#searchButton').on('click', function (e) {
            e.preventDefault();

            var year = parseInt(yearInput.val(), 10);
            var month = monthInput.val();

            if (!location) {
                alert('지역을 먼저 선택해주세요.');
                return;
            }

            if (!isNaN(year) && year >= 1896 && year <= 2051) {
                ajaxRequest(location, year + format02d(month));
            } else {
                alert('범위가 올바르지 않습니다.');
            }
        });
        
        $('#prevMonthBtn').click(function(){
            var currentMm = parseInt($('#monthInput').val());
            var currentYyyy = parseInt($('#yearInput').val());

            if(currentMm == 1){
                currentMm = 12;
                currentYyyy = currentYyyy - 1;
            }else{
                currentMm = currentMm - 1;
            }
            
            $('#yearInput').val(currentYyyy);
            $('#monthInput').val(currentMm);
            $('#searchButton').click();
        });

        $('#nextMonthBtn').click(function(){
            var currentMm = parseInt($('#monthInput').val());
            var currentYyyy = parseInt($('#yearInput').val());

            if(currentMm == 12){
                currentMm = 1;
                currentYyyy = currentYyyy + 1;
            }else{
                currentMm = currentMm + 1;
            }

            $('#yearInput').val(currentYyyy);
            $('#monthInput').val(currentMm);
            $('#searchButton').click();
        });
        
        $('.sunRiseSet').click(function () {
            modalTitle.text($(this).text());
            isMain = $(this).data('main');
            sunRiseSetTableBody.empty();

            if (!iOS) {
                sunRiseSetModal.modal('show');
            } else {
                openSunrisesetWindow(sunRiseSetModal.html(), isMain)
            }
        });

        sunRiseSetModal.on('shown.bs.modal', function() {
            $.ajax({
                url : "/life/sunRiseSet",

                        tr += '<td>' + checkUndefined(format02d(data.SUN_RISE_HH)) + '</td>';
                        tr += '<td>' + checkUndefined(format02d(data.SUN_RISE_MM)) + '</td>';
                        tr += '</tr>';

                        sunRiseSetTableBody.append(tr);
                    });
                }
            });
        });

        var type = "null";
        if (type == 'main') {
            $('.sunRiseSet').trigger('click');
        }
    });
