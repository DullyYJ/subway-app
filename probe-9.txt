		function autoDetectLocation() {
			if (!navigator.geolocation) return;
			navigator.geolocation.getCurrentPosition(
				function(position) {
					var lat = position.coords.latitude;
					var lng = position.coords.longitude;
					getSgisToken(function(accessToken) {
						if (accessToken == null || accessToken === '') {
							applyLocation(lat, lng, '현재 위치');
							return;
						}
						$.ajax({
							url: "https://sgisapi.mods.go.kr/OpenAPI3/addr/rgeocodewgs84.json",
							type: "GET",
							data: { accessToken: accessToken, x_coor: lng, y_coor: lat, addr_type: "20" },
							success: function(res) {
								var address = '현재 위치';
								if (res.result && res.result.length > 0) {
									var r = res.result[0];
									address = r.sido_nm + (r.sgg_nm ? ' ' + r.sgg_nm : '') + (r.emdong_nm ? ' ' + r.emdong_nm : '');
								}
								applyLocation(lat, lng, address);
							},
							error: function() {
								applyLocation(lat, lng, '현재 위치');
							}
						});
					});
				},
				function() {},
				{ timeout: 5000, maximumAge: 60000 }
			);
		}

		function getSgisToken(callback) {
			$.ajax({
				url: "/sgis/accessToken",
				type: "GET",
				dataType: "json",
				success: function(res) {
					callback(res != null && res.accessToken ? res.accessToken : null);
				},
				error: function() {
					callback(null);
				}
			});
		}

    function parseDateParts(value) {
        if (!value) {
            return null;
        }

        var match = value.match(/^(-?\d{1,4})-(\d{2})-(\d{2})$/);

        if (!match) {
            return null;
        }

        return {
            year: parseInt(match[1], 10),
            month: parseInt(match[2], 10),
            day: parseInt(match[3], 10)
        };
    }

    function compareDateParts(a, b) {
        if (a.year !== b.year) {
            return a.year < b.year ? -1 : 1;
        }

        if (a.month !== b.month) {
            return a.month < b.month ? -1 : 1;
        }

        if (a.day !== b.day) {
            return a.day < b.day ? -1 : 1;
        }

        return 0;
    }

    var minDate = parseDateParts(minDateStr);
    var maxDate = parseDateParts(maxDateStr);

    function isDateInRange(year, month, day) {
        var target = {
            year: year,
            month: month,
            day: day
        };

        if (minDate && compareDateParts(target, minDate) < 0) {
            return false;
        }

        if (maxDate && compareDateParts(target, maxDate) > 0) {
            return false;
        }

        return true;
    }
    
    function createDate(year, monthIndex, day) {
        var date = new Date(0);
        date.setHours(0, 0, 0, 0);
        date.setFullYear(year, monthIndex, day);
        return date;
    }

    /* ---- 달력 테이블 렌더 ---- */
    function renderCalendar(year, month) {
        currentYear  = year;
        currentMonth = month;
        buildYearSelect();
        buildMonthSelect();

/*         var firstDay = new Date(year, month - 1, 1).getDay();
        var lastDate = new Date(year, month, 0).getDate();
        var prevLast = new Date(year, month - 1, 0).getDate(); */
        var today = new Date();
        var firstDay = createDate(year, month - 1, 1).getDay();
        var lastDate = getDaysInMonth(year, month);

        var prevYear = month === 1 ? year - 1 : year;
        var prevMonth = month === 1 ? 12 : month - 1;
        var prevLast = getDaysInMonth(prevYear, prevMonth);

        var tbody = popup.querySelector('.krds-dp-table tbody');
        tbody.innerHTML = '';

        var dayCount = 0;
        for (var row = 0; row < 6; row++) {
            var tr = document.createElement('tr');
            for (var col = 0; col < 7; col++) {
                var td  = document.createElement('td');
                var btn = document.createElement('button');
                btn.type = 'button';

                var cellDate, cellYear = year, cellMonth = month, disabled = false;

                if (row === 0 && col < firstDay) {
                    cellDate  = prevLast - firstDay + col + 1;
                    cellYear  = month === 1 ? year - 1 : year;
                    cellMonth = month === 1 ? 12 : month - 1;
                    disabled  = true;
                } else {
                    dayCount++;
                    if (dayCount > lastDate) {
                        cellDate  = dayCount - lastDate;
                        cellYear  = month === 12 ? year + 1 : year;
                        cellMonth = month === 12 ? 1 : month + 1;
                        disabled  = true;
                    } else {
                        cellDate = dayCount;
                    }
                }

                btn.textContent = cellDate;
                
                var outOfRange = !isDateInRange(
                	    cellYear,
                	    cellMonth,
                	    cellDate
                );

                if (disabled || outOfRange) {
                    btn.disabled = true;
                } else {
                    var ds = cellYear + '-'
                           + String(cellMonth).padStart(2,'0') + '-'
                           + String(cellDate).padStart(2,'0');
                    btn.setAttribute('data-date', ds);

                    if (cellYear  === today.getFullYear()  &&
                        cellMonth === today.getMonth() + 1 &&
                        cellDate  === today.getDate()) {
                        btn.classList.add('krds-dp-today');
                    }
                    if (ds === selectedDateStr) {
                        btn.classList.add('krds-dp-selected');
                    }

                    /* 날짜 클릭: 선택만 표시 (확인 버튼으로 최종 반영) */
                    btn.addEventListener('click', (function(dateStr, b) {
                        return function() {
                            selectedDateStr = dateStr;
                            popup.querySelectorAll('.krds-dp-table button').forEach(function(x) {
                                x.classList.remove('krds-dp-selected');
                            });
                            b.classList.add('krds-dp-selected');
                        };
                    })(ds, btn));
                }

                if (col === 0) td.classList.add('krds-dp-sun');
                td.appendChild(btn);
                tr.appendChild(td);
            }
            tbody.appendChild(tr);
            if (dayCount >= lastDate && row >= 4) break;
        }
    }
    
    function isLeapYear(year) {
        return year % 4 === 0
            && (year % 100 !== 0 || year % 400 === 0);
    }

    function getDaysInMonth(year, month) {
        if (month === 2) {
            return isLeapYear(year) ? 29 : 28;
        }

        if (month === 4 || month === 6 || month === 9 || month === 11) {
            return 30;
        }

        return 31;
    }
    
    /* 이전/다음 달 이동 제한 */
    function isMonthInRange(year, month) {
        var firstDay = {
            year: year,
            month: month,
            day: 1
        };

        var lastDay = {
            year: year,
            month: month,
            day: getDaysInMonth(year, month)
        };

        if (minDate && compareDateParts(lastDay, minDate) < 0) {
            return false;
        }

        if (maxDate && compareDateParts(firstDay, maxDate) > 0) {
            return false;
        }

        return true;
    }
