<script type="text/javascript">
(function() {
    /* ---- 대상 요소 (wrapperId 기준으로 스코프 한정 → 여러 개 공존 가능) ---- */
    var wrap    = document.getElementById('dp');
    if (!wrap) return;

    var inp     = wrap.querySelector('#convertSolarLunarDate');
    var popup   = wrap.querySelector('.krds-dp-popup');
    var openBtn = wrap.querySelector('.krds-dp-open-btn');

    var currentYear, currentMonth;
    var selectedDateStr = ''; // yyyy-mm-dd
    
    var minDateStr = '-59-02-13';
    var maxDateStr = '2050-12-31';

    /* ---- 연도 select 생성 (선택된 연도 기준 ±5년) ---- */
    function buildYearSelect() {
        var sel     = popup.querySelector('.krds-dp-year-sel');
        sel.innerHTML = '';
        
        var startYear = minDate ? minDate.year : currentYear - 5;
        var endYear = maxDate ? maxDate.year : currentYear + 5;
        
        for (var y = startYear; y <= endYear; y++) {
            var opt = document.createElement('option');
            opt.value       = y;
            opt.textContent = y + '년';
            if (y === currentYear) opt.selected = true;
            sel.appendChild(opt);
        }
    }

    /* ---- 월 select 생성 ---- */
    function buildMonthSelect() {
        var sel = popup.querySelector('.krds-dp-month-sel');
        sel.innerHTML = '';
        for (var m = 1; m <= 12; m++) {
            var opt = document.createElement('option');
            opt.value       = m;
            opt.textContent = String(m).padStart(2, '0') + '월';
            if (m === currentMonth) opt.selected = true;
            sel.appendChild(opt);
        }
    }
    
    /* 날짜 비교 */
    function parseDateParts(value) {
        if (!value) {
            return null;
        }

        var match = value.match(/^(-?\d{1,4})-(\d{2})-(\d{2})$/);

        if (!match) {
            return null;
        }

        return {
            year: parseInt(match[1], 10),
            month: parseInt(match[2], 10),
            day: parseInt(match[3], 10)
        };
    }

    function compareDateParts(a, b) {
        if (a.year !== b.year) {
            return a.year < b.year ? -1 : 1;
        }

        if (a.month !== b.month) {
            return a.month < b.month ? -1 : 1;
        }

        if (a.day !== b.day) {
            return a.day < b.day ? -1 : 1;
        }

        return 0;
    }

    var minDate = parseDateParts(minDateStr);
    var maxDate = parseDateParts(maxDateStr);

    function isDateInRange(year, month, day) {
        var target = {
            year: year,
            month: month,
            day: day
        };

        if (minDate && compareDateParts(target, minDate) < 0) {
            return false;
        }

        if (maxDate && compareDateParts(target, maxDate) > 0) {
            return false;
        }

        return true;
    }
    
    function createDate(year, monthIndex, day) {
        var date = new Date(0);
        date.setHours(0, 0, 0, 0);
        date.setFullYear(year, monthIndex, day);
        return date;
    }

    /* ---- 달력 테이블 렌더 ---- */
    function renderCalendar(year, month) {
        currentYear  = year;
        currentMonth = month;
        buildYearSelect();
        buildMonthSelect();

/*         var firstDay = new Date(year, month - 1, 1).getDay();
        var lastDate = new Date(year, month, 0).getDate();
        var prevLast = new Date(year, month - 1, 0).getDate(); */
        var today = new Date();
        var firstDay = createDate(year, month - 1, 1).getDay();
        var lastDate = getDaysInMonth(year, month);

        var prevYear = month === 1 ? year - 1 : year;
        var prevMonth = month === 1 ? 12 : month - 1;
        var prevLast = getDaysInMonth(prevYear, prevMonth);

        var tbody = popup.querySelector('.krds-dp-table tbody');
        tbody.innerHTML = '';

        var dayCount = 0;
        for (var row = 0; row < 6; row++) {
            var tr = document.createElement('tr');
            for (var col = 0; col < 7; col++) {
                var td  = document.createElement('td');
                var btn = document.createElement('button');
                btn.type = 'button';

                var cellDate, cellYear = year, cellMonth = month, disabled = false;

                if (row === 0 && col < firstDay) {
                    cellDate  = prevLast - firstDay + col + 1;
                    cellYear  = month === 1 ? year - 1 : year;
                    cellMonth = month === 1 ? 12 : month - 1;
                    disabled  = true;
                } else {
                    dayCount++;
                    if (dayCount > lastDate) {
                        cellDate  = dayCount - lastDate;
                        cellYear  = month === 12 ? year + 1 : year;
                        cellMonth = month === 12 ? 1 : month + 1;
                        disabled  = true;
                    } else {
                        cellDate = dayCount;
                    }
                }

                btn.textContent = cellDate;
                
                var outOfRange = !isDateInRange(
                	    cellYear,
                	    cellMonth,
                	    cellDate
                );

                if (disabled || outOfRange) {
                    btn.disabled = true;
                } else {
                    var ds = cellYear + '-'
                           + String(cellMonth).padStart(2,'0') + '-'
                           + String(cellDate).padStart(2,'0');
                    btn.setAttribute('data-date', ds);

                    if (cellYear  === today.getFullYear()  &&
                        cellMonth === today.getMonth() + 1 &&
                        cellDate  === today.getDate()) {
                        btn.classList.add('krds-dp-today');
                    }
                    if (ds === selectedDateStr) {
                        btn.classList.add('krds-dp-selected');
                    }

                    /* 날짜 클릭: 선택만 표시 (확인 버튼으로 최종 반영) */
                    btn.addEventListener('click', (function(dateStr, b) {
                        return function() {
                            selectedDateStr = dateStr;
                            popup.querySelectorAll('.krds-dp-table button').forEach(function(x) {
                                x.classList.remove('krds-dp-selected');
                            });
                            b.classList.add('krds-dp-selected');
                        };
                    })(ds, btn));
                }

                if (col === 0) td.classList.add('krds-dp-sun');
                td.appendChild(btn);
                tr.appendChild(td);
            }
            tbody.appendChild(tr);
            if (dayCount >= lastDate && row >= 4) break;
        }
    }
    
    function isLeapYear(year) {
        return year % 4 === 0
            && (year % 100 !== 0 || year % 400 === 0);
    }

    function getDaysInMonth(year, month) {
        if (month === 2) {
            return isLeapYear(year) ? 29 : 28;
        }

        if (month === 4 || month === 6 || month === 9 || month === 11) {
            return 30;
        }

        return 31;
    }
    
    /* 이전/다음 달 이동 제한 */
    function isMonthInRange(year, month) {
        var firstDay = {
            year: year,
            month: month,
            day: 1
        };

        var lastDay = {
            year: year,
            month: month,
            day: getDaysInMonth(year, month)
        };

        if (minDate && compareDateParts(lastDay, minDate) < 0) {
            return false;
        }

        if (maxDate && compareDateParts(firstDay, maxDate) > 0) {
            return false;
        }

        return true;
    }

    /* ---- 팝업 열기 ---- */
/*     function openPopup() {
        var val = inp.value;
        if (val && /^\d{4}-\d{2}-\d{2}$/.test(val)) {
            var p = val.split('-');
            currentYear     = parseInt(p[0]);
            currentMonth    = parseInt(p[1]);
            selectedDateStr = val;
        } else {
            var t = new Date();
            currentYear     = t.getFullYear();
            currentMonth    = t.getMonth() + 1;
            selectedDateStr = '';
        }
        renderCalendar(currentYear, currentMonth);
        popup.classList.add('open');
    } */
    function openPopup() {
        var val = inp.value;
        var parsed = parseDateParts(val);

        if (parsed && isDateInRange(parsed.year, parsed.month, parsed.day)) {
            currentYear = parsed.year;
            currentMonth = parsed.month;
            selectedDateStr = val;
        } else {
            var t = new Date();
            currentYear = t.getFullYear();
            currentMonth = t.getMonth() + 1;
            selectedDateStr = '';
        }

        renderCalendar(currentYear, currentMonth);
        popup.classList.add('open');
    }

    /* ---- 팝업 닫기 ---- */
    function closePopup() {
        popup.classList.remove('open');
    }

    /* ---- 이벤트 바인딩 ---- */
    // 달력 아이콘 버튼
    openBtn.addEventListener('click', function(e) {
        e.stopPropagation();
        popup.classList.contains('open') ? closePopup() : openPopup();
    });

    // input 클릭으로도 열기
    inp.addEventListener('click', function(e) {
        e.stopPropagation();
        if (!popup.classList.contains('open')) openPopup();
    });

    // 외부 클릭 시 닫기
    document.addEventListener('click', function(e) {
        if (!wrap.contains(e.target)) closePopup();
    });

    // 이전 달
/*     popup.querySelector('.krds-dp-prev').addEventListener('click', function() {
        var m = currentMonth - 1, y = currentYear;
        if (m < 1) { m = 12; y--; }
        renderCalendar(y, m);
    }); */
    popup.querySelector('.krds-dp-prev').addEventListener('click', function() {
        var m = currentMonth - 1;
        var y = currentYear;

        if (m < 1) {
            m = 12;
            y--;
        }

        if (isMonthInRange(y, m)) {
            renderCalendar(y, m);
        }
    });

    // 다음 달
/*     popup.querySelector('.krds-dp-next').addEventListener('click', function() {
        var m = currentMonth + 1, y = currentYear;
        if (m > 12) { m = 1; y++; }
        renderCalendar(y, m);
    }); */
    popup.querySelector('.krds-dp-next').addEventListener('click', function() {
        var m = currentMonth + 1;
        var y = currentYear;

        if (m > 12) {
            m = 1;
            y++;
        }

        if (isMonthInRange(y, m)) {
            renderCalendar(y, m);
        }
    });

/*     // 연도 select
    popup.querySelector('.krds-dp-year-sel').addEventListener('change', function() {
        renderCalendar(parseInt(this.value), currentMonth);
    });

    // 월 select
    popup.querySelector('.krds-dp-month-sel').addEventListener('change', function() {
        renderCalendar(currentYear, parseInt(this.value));
    }); */
    popup.querySelector('.krds-dp-year-sel').addEventListener('change', function() {
        var year = parseInt(this.value, 10);

        if (isMonthInRange(year, currentMonth)) {
            renderCalendar(year, currentMonth);
        }
    });

    popup.querySelector('.krds-dp-month-sel').addEventListener('change', function() {
        var month = parseInt(this.value, 10);

        if (isMonthInRange(currentYear, month)) {
            renderCalendar(currentYear, month);
        }
    });

    // 오늘 버튼
    popup.querySelector('.krds-dp-btn-today').addEventListener('click', function() {
        var t = new Date();
        selectedDateStr = t.getFullYear() + '-'
                        + String(t.getMonth()+1).padStart(2,'0') + '-'
                        + String(t.getDate()).padStart(2,'0');
        renderCalendar(t.getFullYear(), t.getMonth() + 1);
    });

    // 취소 버튼
    popup.querySelector('.krds-dp-btn-cancel').addEventListener('click', function() {
        closePopup();
    });

    // 확인 버튼: input에 값 반영 후 닫기
    popup.querySelector('.krds-dp-btn-confirm').addEventListener('click', function() {
        if (selectedDateStr) {
            inp.value = selectedDateStr;
            /* name이 다른 hidden input이 별도로 있는 경우도 동기화 */
            var form = inp.closest('form');
            if (form) {
                var hidden = form.querySelector('input[name="date"][type="hidden"]');
                if (hidden) hidden.value = selectedDateStr;
            }
        }
        closePopup();
    });

})();
</script>
