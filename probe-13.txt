		function fn_convert() {
			if($('#mp').is(':visible')){	//달력
			
			
				if($('#mp-input').val().split('-').length >= 3){
					$("#search_year").val("-" + $('#mp-input').val().split('-')[1])
					$("#search_month").val($('#mp-input').val().split('-')[2])
				} else{
					$("#search_year").val($('#mp-input').val().split('-')[0])
					$("#search_month").val($('#mp-input').val().split('-')[1])
				}
				
				if(!$("input:checkbox[id='dp_11']").is(":checked") && !$("input:checkbox[id='dp_12']").is(":checked")){
					alert("검색하실 역법을 선택해주세요.")
					return;
				}
				
				if($("input:checkbox[id='dp_11']").is(":checked") && $("input:checkbox[id='dp_12']").is(":checked")){
					alert("하나의 역법을 선택해주세요.")
					return;
				}
				
				if($("input:checkbox[id='dp_11']").is(":checked")){
					$("#search_check").val("J");
				} else{
					$("#search_check").val("G");
				}
				
				$("#search_dp").val("0");
				$('#convertForm').submit();
				
			}else {	//입력폼	
				if ( !isNumber($('#year01').val()) || !isNumber($('#month01').val()) ) {
					alert('입력된 연 또는 월이 숫자형태가 아닙니다.');
					return;
				}
				if ( $('#year01').val() < -59 ) {
					alert('양력 -59년 이후의 날짜에 대해서만 음양력변환이 가능합니다.');
					return;
				}
				if ( $('#year01').val() == -59 && $('#month01').val() == 1 ) {
					alert('양력 -59년 2월 이후의 날짜에 대해서만 음양력변환이 가능합니다.');
					return;
				}
				if ( $('#year01').val() > 2050 ) {
					alert('양력 2050년 12월까지만 음양력변환이 가능합니다.');
					return;
				}
				if ( $('#month01').val() > 12 ) {
					alert('월은 12보다 작아야 합니다.');
					return;
				}
				$("#search_year").val($('#year01').val())

    function parseYearMonth(value) {
        if (!value) {
            return null;
        }

        var match = value.match(/^(-?\d{1,4})-(\d{2})$/);

        if (!match) {
            return null;
        }

        return {

	    var startYear = minMonthValue
	        ? minMonthValue.year
	        : currentYear - 5;
	
	    var endYear = maxMonthValue
	        ? maxMonthValue.year
	        : currentYear + 5;
	
	    for (var year = startYear; year <= endYear; year++) {
	        var option = document.createElement('option');
	
	        option.value = year;
	        option.textContent = year + '년';
	        option.selected = year === currentYear;

    function isMonthInRange(year, monthIndex) {
        var month = monthIndex + 1;

                return false;

            if (
                year === minMonthValue.year &&
                month < minMonthValue.month
            ) {
                return false;
            }
        }

                return false;
            }

            if (
                year === maxMonthValue.year &&
                month > maxMonthValue.month
            ) {
                return false;
            }
        }

        return true;

    function openPopup() {
        var parsed = parseYearMonth(inp.value);

        if (
            parsed &&
            isMonthInRange(parsed.year, parsed.month - 1)
        ) {
            currentYear = parsed.year;
            selectedYear = parsed.year;
            selectedMonth = parsed.month - 1;
        } else {
            var t = new Date();

            currentYear = t.getFullYear();
            selectedYear = null;
            selectedMonth = null;

        tempYear = selectedYear;
        tempMonth = selectedMonth;

        renderGrid();
        popup.classList.add('open');

    function applySelection() {
        if (tempYear === null || tempMonth === null) return;
        selectedYear  = tempYear;
        selectedMonth = tempMonth;
        var formatted = selectedYear + '-' + String(selectedMonth + 1).padStart(2, '0');
        inp.value = formatted;

        /* hidden input 동기화 (기존 krdsCalendar.jsp와 동일 패턴) */
        var form = inp.closest('form');
        if (form) {
            var hidden = form.querySelector('input[name="date"][type="hidden"]');
            if (hidden) hidden.value = formatted;
        }

        /* 커스텀 이벤트 발행 → 외부에서 listen 가능 */
        var ev = new CustomEvent('krdsMonthChange', {
            bubbles: true,
            detail: { year: selectedYear, month: selectedMonth + 1, formatted: formatted }
        });
        inp.dispatchEvent(ev);

        closePopup();
    }
